/**
 * Parsea una fecha de forma segura para comparar "día calendario" en
 * Argentina. "YYYY-MM-DD" sin hora se interpreta como medianoche UTC,
 * que en Argentina (UTC-3) cae en el día ANTERIOR — bug real encontrado
 * en getArqueoStatus el 13/08/2026 (una fecha pedida como "2026-08-07"
 * devolvía datos del 06/08). Si el string ya trae hora, se respeta tal cual.
 */
function parseFechaSegura(dateStr) {
  if (typeof dateStr === 'string' && /^\d{4}-\d{2}-\d{2}$/.test(dateStr)) {
    return new Date(dateStr + "T12:00:00");
  }
  return new Date(dateStr);
}

/****************************************************
 * 🏠 LÓGICA DE GASTOS FIJOS
 ****************************************************/
function getConfigGastosFijos() {
  const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAMES.CONFIG_GASTOS_FIJOS);
  if (!sh) return { ok: false, error: 'Hoja ConfigGastosFijos no inicializada' };
  
  const data = sh.getDataRange().getValues();
  const res = [];
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (row[0]) { 
      res.push({ 
        nombre: row[0], 
        montoMensual: Number(row[1]) || 0,
        lastUpdate: row[2] instanceof Date ? Utilities.formatDate(row[2], "America/Argentina/Buenos_Aires", "yyyy-MM-dd") : ''
      });
    }
  }
  return { ok: true, data: res };
}

function addConfigGastoFijo(nombre, montoMensual) {
  const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAMES.CONFIG_GASTOS_FIJOS);
  if (!sh) return { ok: false, error: 'Hoja no existe' };
  
  // Opcional: Chequeo de duplicados
  const data = sh.getDataRange().getValues();
  for(let i=1; i<data.length; i++) {
    if(String(data[i][0]).toLowerCase() === String(nombre).toLowerCase()) {
      return { ok: false, error: 'El gasto fijo ya existe.' };
    }
  }
  sh.appendRow([nombre, Number(montoMensual), new Date()]);
  return { ok: true };
}

function updateConfigGastoFijo(nombre, montoMensual, lastUpdate) {
  const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAMES.CONFIG_GASTOS_FIJOS);
  if (!sh) return { ok: false, error: 'Hoja no existe' };

  const data = sh.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).toLowerCase() === String(nombre).toLowerCase()) {
      sh.getRange(i + 1, 2).setValue(Number(montoMensual));
      // Actualiza la columna FechaUltimaActualizacion (Col 3)
      sh.getRange(i + 1, 3).setValue(new Date(lastUpdate)); 
      return { ok: true };
    }
  }
  return { ok: false, error: 'Gasto fijo no encontrado' };
}

/****************************************************
 * 🤖 JOB ORQUESTADOR V4.2 (CICLO STRICTO)
 * --------------------------------------------------
 * Lógica: Barrido Cronológico Ajustado a Laura.
 * * 1. PRIORIDAD CRÍTICA: Ayer Mañana (Recupero de día agitado)
 * 2. PRIORIDAD ALTA: Ayer Tarde (Lo estándar al iniciar el día)
 * 3. PRIORIDAD NORMAL: Hoy Mañana (Lo que llega al mediodía)
 * * NOTA: "Hoy Tarde" NO se busca. Se procesará mañana como "Ayer Tarde".
 ****************************************************/
function JOB_OrquestadorRendiciones() {
  const ss = SpreadsheetApp.getActive();
  const cacheSh = ss.getSheetByName(SHEET_NAMES.CACHE);
  
  let histSh = ss.getSheetByName(SHEET_NAMES.HISTORIAL_IMPORTS);
  if (!histSh) { histSh = ss.insertSheet(SHEET_NAMES.HISTORIAL_IMPORTS); histSh.appendRow(['Fecha','FileID','Nombre']); }

  // 🛑 FRENO DE MANO: Si Laura ya tiene algo en pantalla, no la molestamos.
  if (cacheSh.getLastRow() > 1) {
    console.log("⏳ JOB: Hay una rendición pendiente en pantalla. Esperando al usuario.");
    return;
  }
  
  const hoy = new Date();
  const ayer = new Date(hoy.getTime() - 24 * 60 * 60 * 1000);
  
  console.log("🕵️ JOB: Iniciando escaneo (V4.2 - Sin adelantarse al futuro)...");

  // =================================================================
  // 1️⃣ EL RECUPERO: AYER - TURNO MAÑANA
  // =================================================================
  // Si ayer fue un caos y Nico no subió la mañana, la sube hoy.
  // Esto debe aparecer ANTES que la tarde de ayer para respetar el orden de la caja.
  const ayerManana = buscarArchivoInteligente(ayer, "Mañana", REPARTIDOR_PRINCIPAL);
  if (ayerManana && !yaFueImportado(ayerManana.id, histSh)) {
    console.log(`🚨 RECUPERADO: Turno Mañana de AYER (${ayerManana.fecha})`);
    
    cargarEnCache(cacheSh, histSh, ayerManana);
    
    registrarNotificacion(
      'INFO', 'ALTA', 
      'Rendición Recuperada', 
      `Apareció el turno Mañana de AYER que estaba pendiente.`, 
      ayerManana
    );
    return; // Cortamos y entregamos esto.
  }

  // =================================================================
  // 2️⃣ EL CIERRE: AYER - TURNO TARDE
  // =================================================================
  // Esto es lo primero que Laura debería ver en un día normal al llegar.
  const ayerTarde = buscarArchivoInteligente(ayer, "Tarde", REPARTIDOR_PRINCIPAL);
  if (ayerTarde && !yaFueImportado(ayerTarde.id, histSh)) {
    console.log(`✅ ENCONTRADO: Turno Tarde de AYER (${ayerTarde.fecha})`);
    cargarEnCache(cacheSh, histSh, ayerTarde);
    // Notificación normal
    registrarNotificacion('INFO', 'MEDIA', 'Rendición Pendiente', `Listo el cierre de ayer tarde.`, ayerTarde);
    return;
  }

  // =================================================================
  // 3️⃣ EL DÍA EN CURSO: HOY - TURNO MAÑANA
  // =================================================================
  // Esto llega después del mediodía (13:00 - 15:00 hs).
  const hoyManana = buscarArchivoInteligente(hoy, "Mañana", REPARTIDOR_PRINCIPAL);
  if (hoyManana && !yaFueImportado(hoyManana.id, histSh)) {
    console.log(`✅ ENCONTRADO: Turno Mañana de HOY (${hoyManana.fecha})`);
    cargarEnCache(cacheSh, histSh, hoyManana);
    registrarNotificacion('INFO', 'BAJA', 'Nueva Rendición', `Llegó el turno mañana de hoy.`, hoyManana);
    return;
  }
  
  // =================================================================
  // 4️⃣ EL CIERRE DE HOY: HOY - TURNO TARDE
  // =================================================================
  const hoyTarde = buscarArchivoInteligente(hoy, "Tarde", REPARTIDOR_PRINCIPAL);
  if (hoyTarde && !yaFueImportado(hoyTarde.id, histSh)) {
    console.log(`✅ ENCONTRADO: Turno Tarde de HOY (${hoyTarde.fecha})`);
    cargarEnCache(cacheSh, histSh, hoyTarde);
    registrarNotificacion('INFO', 'MEDIA', 'Cierre de Hoy', `Listo el cierre de hoy a la tarde.`, hoyTarde);
    return;
  }

  console.log("💤 JOB: Nada pendiente por ahora.");
}


// --- Helper de Búsqueda Inteligente (Anti-PDF + Lector GSheet + Búsqueda Root) ---
function buscarArchivoInteligente(fecha, turno, repartidor) {
  if (!fecha || !(fecha instanceof Date)) return null;

  const fechaStr = Utilities.formatDate(fecha, "America/Argentina/Buenos_Aires", "yyyy-MM-dd");
  
  // ⚠️ CAMBIO CRÍTICO: Ya no buscamos el nombre exacto del repartidor ("Nico").
  // Ahora buscamos coincidencia FLEXIBLE: Que tenga "Rendición" + TURNO + FECHA.
  // Esto soluciona problemas de tildes ("Nicolás" vs "Nico") o cambios de nombre.
  
  try {
    const rootFolder = DriveApp.getFolderById(RECIBOS_FOLDER_ID);
    
    // Nombres de carpeta posibles
    const meses = ["Enero","Febrero","Marzo","Abril","Mayo","Junio","Julio","Agosto","Septiembre","Octubre","Noviembre","Diciembre"];
    const mesIndex = parseInt(Utilities.formatDate(fecha, "America/Argentina/Buenos_Aires", "M")) - 1;
    const anio = Utilities.formatDate(fecha, "America/Argentina/Buenos_Aires", "yyyy");
    
    const nombreCarpetaEs = `${meses[mesIndex]} ${anio}`; // Enero 2026
    const nombreCarpetaEn = Utilities.formatDate(fecha, "en", "MMMM yyyy"); // January 2026

    // 1. Recopilamos DÓNDE buscar (Carpetas mes + Root)
    let carpetasABuscar = [rootFolder]; // Siempre buscamos en la raíz por si acaso

    let subFoldersEs = rootFolder.getFoldersByName(nombreCarpetaEs);
    if (subFoldersEs.hasNext()) carpetasABuscar.push(subFoldersEs.next());
    
    let subFoldersEn = rootFolder.getFoldersByName(nombreCarpetaEn);
    if (subFoldersEn.hasNext()) carpetasABuscar.push(subFoldersEn.next());

    // 2. Buscamos en todas las carpetas candidatas
    for (let folder of carpetasABuscar) {
        const files = folder.getFiles();
        while (files.hasNext()) {
            const file = files.next();
            
            // Filtros rápidos de basura
            if (file.isTrashed()) continue;
            if (file.getMimeType() === MimeType.PDF) continue;

            const nombreArchivo = file.getName();

            // 🔥 LÓGICA FLEXIBLE:
            // 1. Tiene la fecha (2026-01-09)?
            // 2. Tiene el turno (Mañana/Tarde)?
            // 3. Dice "Rendición"?
            if (nombreArchivo.includes(fechaStr) && nombreArchivo.includes(turno) && nombreArchivo.includes("Rendición")) {
                
                // ¡EUREKA! Encontramos el archivo (sin importar el nombre del chofer)
                try {
                    let totalMonto = 0;
                    let jsonString = "{}";

                    if (file.getMimeType() === MimeType.GOOGLE_SHEETS) {
                        console.log(`📊 Leyendo Planilla: ${file.getName()}`);
                        totalMonto = getTotalEfectivoDesdePlanillaLegacy(file.getId());
                        jsonString = JSON.stringify({ 
                            origen: "GOOGLE_SHEET", fecha: fechaStr, turno: turno, 
                            repartidor: repartidor, efectivoEsperado: totalMonto
                        }); 
                        if (totalMonto === 0) console.warn("⚠️ OJO: El monto leído es $0.");
                    }
                    else if (file.getName().endsWith(".json") || file.getMimeType() === MimeType.PLAIN_TEXT) {
                        let contenido = file.getBlob().getDataAsString();
                        if (contenido.trim().startsWith("{")) {
                            const json = JSON.parse(contenido);
                            totalMonto = json.totalEfectivo || 0;
                            jsonString = contenido;
                        }
                    }

                    return {
                        id: file.getId(),
                        name: file.getName(),
                        fecha: fechaStr,
                        turno: turno,
                        repartidor: repartidor,
                        monto: totalMonto,
                        json: jsonString
                    };

                } catch(readError) {
                    console.log(`❌ Error leyendo archivo "${file.getName()}": ${readError.message}`);
                }
            }
        }
    }

  } catch(e) {
    console.log("❌ Error general buscando archivo: " + e.message);
  }
  return null;
}

// VERSIÓN V7.0 - "SCANNER DE ZONA" (Busca el número mayor en el área)
function getTotalEfectivoDesdePlanillaLegacy(fileId) {
  try {
    const ss = SpreadsheetApp.openById(fileId);
    const sh = ss.getSheets()[0];
    
    // Palabras clave
    const keywords = ["TOTAL EFECTIVO", "TOTAL RECIBIDO", "A RENDIR", "EFECTIVO A RENDIR"];
    let match = null;
    let labelFound = "";

    // 1. Encontrar fila de la etiqueta
    for (let word of keywords) {
      match = sh.createTextFinder(word).matchCase(false).findNext();
      if (match) {
        labelFound = word;
        break; 
      }
    }
    
    if (match) {
       const row = match.getRow();
       const col = match.getColumn();
       
       // 2. ESCANEO AMPLIO (ZONA DE CALOR)
       // Leemos 2 filas (la de la etiqueta y la de abajo)
       // Leemos 10 columnas a la derecha (para cubrir cualquier desplazamiento)
       // Range: desde [row, col] hasta [row+1, col+10]
       const zonaDatos = sh.getRange(row, col + 1, 2, 10).getValues();
       
       console.log(`🧐 Scanner V7 (${fileId}): Etiqueta "${labelFound}" en fila ${row}. Escaneando zona...`);

       let maxVal = 0;
       
       // Función Limpieza
       const limpiarValor = (v) => {
          if (typeof v === 'number') return v;
          if (!v) return 0;
          if (typeof v === 'string') {
             let s = v.replace(/\$/g, '').trim();
             if (s === "") return 0;
             // Formato Argentina: 1.500,50 -> 1500.50
             if (s.includes(',')) s = s.replace(/\./g, '').replace(',', '.');
             else if (s.includes('.')) s = s.replace(/\./g, ''); // Si solo tiene puntos, son miles
             return parseFloat(s); 
          }
          return 0;
       };

       // 3. BUSCAR EL NÚMERO MÁS GRANDE EN LA ZONA
       // El Total casi siempre es el número más grande de esa sección final.
       zonaDatos.forEach(fila => {
           fila.forEach(celda => {
               const valor = limpiarValor(celda);
               if (!isNaN(valor) && valor > maxVal) {
                   maxVal = valor;
               }
           });
       });

       if (maxVal === 0) {
          console.warn(`⚠️ ALERTA: La etiqueta está, pero no hay ningún número mayor a 0 en las 10 celdas siguientes ni abajo.`);
       } else {
          console.log(`✅ EXCELENTE: Monto detectado: $ ${maxVal}`);
       }
       
       return maxVal;
    } else {
       console.log("⚠️ No se encontró la etiqueta 'TOTAL EFECTIVO'.");
    }
  } catch(e) {
    console.log("❌ Error crítico: " + e.message);
  }
  return 0;
}

function yaFueImportado(fileId, sheetHist) {
  if (sheetHist.getLastRow() < 2) return false;
  const data = sheetHist.getRange(2, 2, sheetHist.getLastRow()-1, 1).getValues();
  const listaIds = data.flat().map(String);
  return listaIds.includes(String(fileId));
}

    };
  }

  const sbRendicion = buscarUltimaRendicionPendienteSupabase_();
  if (sbRendicion) {
    Logger.log('✅ Rendición rescatada desde Supabase (cache vacía).');
    return {
      ok: true,

function procesarRendicionDesdeRecibo(p) {
  if (yaFueRendidaEnHoja(p.fechaStr, p.turno, p.repartidor)) {
    borrarCache();
    marcarRendicionProcesadaEnSupabase_(p.fechaStr, p.turno, p.repartidor);
    return { ok: true, mensaje: "Planilla ya estaba registrada. Limpiando..." };
  }

  borrarCache();
  marcarRendicionProcesadaEnSupabase_(p.fechaStr, p.turno, p.repartidor);
  return registro;
}


function yaFueRendidaEnHoja(fechaStr, turno, repartidor) {
  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName(SHEET_NAMES.RENDICIONES);
  if (!sh || sh.getLastRow() < 2) return false;

  const fBusq = Utilities.formatDate(parseFechaSegura(fechaStr), "America/Argentina/Buenos_Aires", "dd/MM/yyyy");
  const data = sh.getDataRange().getValues(); 

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (!row[1]) continue;
    const fRow = Utilities.formatDate(new Date(row[1]), "America/Argentina/Buenos_Aires", "dd/MM/yyyy");
    if (fRow === fBusq && String(row[2]) === String(turno) && String(row[3]) === String(repartidor)) {
      return true;
    }
  }
  return false;
}

// 🔥 FUNCIÓN CORE DE CONTABILIDAD CORREGIDA (V3.8)
function registrarRendicionEnHoja(p) {
  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName(SHEET_NAMES.RENDICIONES);
  
  const efectivoEsperado = Number(p.efectivoEsperado);
  const efectivoContado = Number(p.efectivoContado);
  const diferencia = efectivoContado - efectivoEsperado;
  
  let tipoDif = 'Exacto';
  if (diferencia > 0) tipoDif = 'Sobrante';
  if (diferencia < 0) tipoDif = 'Faltante';

  analizarComportamiento(p.monitorData, "RENDICION", p.usuario, diferencia);

  const nextId = sh.getLastRow() < 2 ? 1 : Number(sh.getRange(sh.getLastRow(), 1).getValue()) + 1;
  const now = new Date();

  // A. Registrar en Hoja Rendiciones (para historial y diferencias)
  sh.appendRow([
    nextId, parseFechaSegura(p.fechaStr), p.turno, p.repartidor, efectivoEsperado, efectivoContado,
    diferencia, tipoDif, p.usuario, now, p.fileId, JSON.stringify(p.monitorData || {}) 
  ]);

  // 🔥 B. MOVIMIENTO 1: Registramos la BASE ESPERADA como INGRESO
  // Este movimiento refleja la parte de la plata que sí debería haber entrado.
  registrarMovimientoCaja(
    'Ingreso', 'Efectivo', efectivoEsperado, 'Rendición Reparto - BASE', 
    p.repartidor, p.turno, '', '', p.usuario, `Base Rendición ${p.repartidor} (${p.turno})`, "", "", ""
  );

  // 🔥 C. MOVIMIENTO 2: Registramos el AJUSTE (sobrante/faltante)
  // Este movimiento cuadra la caja al monto real que entró (Contado = Esperado + Diferencia).
  if (diferencia !== 0) {
    const tipoMov = diferencia > 0 ? 'Ingreso' : 'Egreso'; // Sobrante es Ingreso, Faltante es Egreso
    registrarMovimientoCaja(
      tipoMov, 'Efectivo', Math.abs(diferencia), 'Diferencia Rendición - Ajuste', 
      p.repartidor, p.turno, '', '', p.usuario, `Ajuste automático ${tipoDif}`
    );
  }
  
  // Resultado Neto en Caja: Ingreso(Esperado) + Ingreso/Egreso(Diferencia) = Contado. (Correcto)

  logAccion(p.usuario, 'RENDICION', `Dif: ${diferencia}`, `ID:${nextId}`);

  return { ok: true, diferencia };
}

function analizarComportamiento(data, tipoEvento, usuario, diferencia) {
  if (!data) return;
  const edits = data.edits || 0;
  const deletes = data.deletes || 0;
  const alertas = [];

  if (diferencia === 0 && edits > 15) {
    alertas.push(`POSIBLE DIBUJO: Diferencia exacta ($0) tras ${edits} ediciones de billetes.`);
  }
  if (deletes > 3) {
    alertas.push(`INDECISIÓN: Borró el conteo completo ${deletes} veces.`);
  }
  if (data.submitAttempts && data.submitAttempts > 2) {
    alertas.push(`INTENTOS FALLIDOS: Trató de confirmar ${data.submitAttempts} veces.`);
  }

  if (alertas.length > 0) {
    registrarNotificacion("ANOMALIA_CAJA", "ALTA", `Sospecha en ${tipoEvento}`, `Usuario ${usuario}: ${alertas.join(" ")}`, data);
  }
}

function registrarMovimientoCaja(tipo, forma, importe, cat, rep, turno, banco, ch, user, obs, prov, veh, emp, idChequeOrigen) {
  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName(SHEET_NAMES.MOVIMIENTOS);
  const id = sh.getLastRow() < 2 ? 1000 : Number(sh.getRange(sh.getLastRow(), 1).getValue()) + 1;
  const now = new Date();
  
  if (tipo === 'Egreso' && forma === 'Cheque' && idChequeOrigen) {
    marcarChequeEntregado(idChequeOrigen, now);
  }

  let obsFinal = obs || "";
  // Guards anti-duplicado: la app ya puede mandar el tag dentro de la observación
  if (prov && !obsFinal.includes('[Prov:')) obsFinal = `[Prov: ${prov}] ` + obsFinal;
  if (veh && !obsFinal.includes('[Veh:'))   obsFinal = `[Veh: ${veh}] ` + obsFinal;
  if (emp && !obsFinal.includes(`[${emp}]`)) obsFinal = `[${emp}] ` + obsFinal;

  sh.appendRow([
      id, now, Utilities.formatDate(now, "America/Argentina/Buenos_Aires", "HH:mm"), 
      tipo, forma, banco||"", ch||"", Number(importe), cat, rep||"", turno||"", user||"Sistema", obsFinal, ""
  ]);
  
  logAccion(user, 'MOVIMIENTO_CAJA', `Tipo: ${tipo} - Importe: ${importe}`, cat);

  // 🔥 NUEVO: Verificar alerta de combustible (solo si es egreso de combustible y hay vehículo)
  if (tipo === 'Egreso' && cat.includes('Combustible') && veh) {
    const loads = getCombustibleLoadsThisWeek(veh);
    // Si loads es 2, significa que ya hubo 2 cargas. La actual es la 3ra o más.
    if (loads >= 2) { 
      registrarNotificacion(
        "ALERTA_COMBUSTIBLE", 
        "MEDIA", 
        `ALERTA: 3RA Carga de ${veh}`, 
        `El vehículo ${veh} ya registró ${loads + 1} cargas esta semana. Esto excede el límite normal de 2.`,
        { vehiculo: veh, cargas: loads + 1 }
      );
    }
  }

  // El tablero de combustible (LogsCombustible) YA NO se actualiza acá.
  // Medido en producción: el agregado incremental por fila igual sumaba
  // 5-6 llamadas extra a Sheets (insertRowBefore + setValues + formatos),
  // y cada llamada tiene latencia propia en Apps Script — el registro de
  // Combustible seguía tardando el doble que cualquier otra categoría.
  // Ahora se actualiza solo, cada 10 min, vía trigger (ver
  // JOB_ActualizarTableroCombustiblePeriodico) — así una carga de
  // combustible cuesta exactamente lo mismo que cualquier otro egreso.

  return id;
}

// 🔥 NUEVO: Lógica para contar cargas de combustible en la semana
function getCombustibleLoadsThisWeek(vehiculo) {
  if (!vehiculo) return 0;
  
  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName(SHEET_NAMES.MOVIMIENTOS);
  if (!sh || sh.getLastRow() < 2) return 0;
  
  const hoy = new Date();
  const diaSemana = hoy.getDay(); 
  // Calcular el Lunes de esta semana (inicio de semana: 1=Lunes, 0=Domingo)
  const diffLunes = hoy.getDate() - diaSemana + (diaSemana == 0 ? -6 : 1);
  const lunes = new Date(hoy);
  lunes.setDate(diffLunes);
  lunes.setHours(0, 0, 0, 0); // Inicio del día

  // Optimización: en vez de leer TODA la hoja (miles de filas) alcanza con
  // mirar las últimas ~400 — de sobra para cubrir una semana de actividad
  // normal, y evita releer todo el historial en cada carga de combustible.
  const lastRow  = sh.getLastRow();
  const VENTANA  = 400;
  const startRow = Math.max(2, lastRow - VENTANA + 1);
  const numRows  = lastRow - startRow + 1;
  const data = sh.getRange(startRow, 1, numRows, 13).getValues();
  let count = 0;
  const vehiculoClean = String(vehiculo).toLowerCase();
  
  for (let row of data) {
    const fecha = row[1];
    const tipo = row[3];
    const categoria = String(row[8] || "").toLowerCase();
    const observacion = String(row[12] || "");
    
    if (fecha instanceof Date && fecha >= lunes && fecha <= hoy && tipo === 'Egreso' && categoria.includes("combustible")) {
      const match = observacion.match(/\[Veh: (.*?)\]/i);
      // Usamos vehiculo si viene en la obs, o la categoria si es específica
      let vehiculoEnMovimiento = "";
      if (match) {
        vehiculoEnMovimiento = String(match[1]).toLowerCase();
      } else {
         // Intentar deducir de la categoría si no está en la observación
         if (categoria.includes('saveiro')) vehiculoEnMovimiento = 'saveiro';
         else if (categoria.includes('toyota') || categoria.includes('hiace')) vehiculoEnMovimiento = 'toyota hiace';
         else if (categoria.includes('fiat') || categoria.includes('uno')) vehiculoEnMovimiento = 'fiat uno cargo';
      }
      
      if (vehiculoEnMovimiento === vehiculoClean) {
          count++;
      }
    }
  }
  return count;
}

// 🔥 NUEVO: API para que el frontend pueda chequear la advertencia de combustible
function checkCombustibleWarning(vehiculo) {
  if (!vehiculo) return { ok: false, message: "Vehículo no especificado." };
  const loads = getCombustibleLoadsThisWeek(vehiculo);
  // Si loads es 2, significa que ya hubo 2 cargas. La actual es la 3ra o más.
  return { 
    ok: true, 
    currentLoads: loads, 
    isWarning: loads >= 2,
    message: loads >= 2 ? `¡ALERTA! Esta sería la carga número ${loads + 1} del ${vehiculo} esta semana (límite normal 2).` : null
  };
}

// ✅ NUEVO: Función para obtener estado de arqueo (Dueño)
function getArqueoStatus(dateStr) {
  const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAMES.ARQUEOS);
  if (!sh || sh.getLastRow() < 2) return { status: 'missing', difference: 0 };

  const fBusq = Utilities.formatDate(parseFechaSegura(dateStr), "America/Argentina/Buenos_Aires", "dd/MM/yyyy");
  const data = sh.getDataRange().getValues();

  for (let i = data.length - 1; i >= 1; i--) {
    const row = data[i];
    const fRow = Utilities.formatDate(new Date(row[1]), "America/Argentina/Buenos_Aires", "dd/MM/yyyy");

    if (fRow === fBusq) {
      const difference = Number(row[5]);
      const result = String(row[6]);
      const closeTime = Utilities.formatDate(new Date(row[7]), "America/Argentina/Buenos_Aires", "HH:mm");

      if (result === 'OK') {
        return { status: 'ok', difference: difference, closeTime: closeTime };
      } else if (result === 'Sobrante' || result === 'Faltante') {
        return { status: 'discrepancy', difference: difference, closeTime: closeTime };
      }
    }
  }

  return { status: 'missing', difference: 0 };
}

function getEstadoCaja() {
  const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAMES.MOVIMIENTOS);
  const data = sh.getDataRange().getValues();
  let ef = 0, ch = 0, bc = 0;
  
  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    if (!row[0]) continue;
    
    // row[7] es Importe, row[3] es Tipo (Ingreso/Egreso), row[4] es Forma de Pago
    const importe = Number(row[7]) || 0;
    const esIngreso = row[3] === 'Ingreso';

    // 1. EFECTIVO: Suma ingresos, resta egresos (Correcto)
    if (/efectivo/i.test(row[4])) {
      ef += esIngreso ? importe : -importe;
    } 
    // 2. CHEQUES (CORREGIDO): 
    // Solo sumamos los cheques que entraron y NO han sido entregados.
    // NO restamos los egresos de cheques, porque al registrar el egreso, el sistema marca 
    // el cheque original como "ENTREGADO", sacándolo automáticamente de esta suma.
    else if (/cheque/i.test(row[4])) {
      if (esIngreso && !String(row[13]).includes("ENTREGADO")) {
        ch += importe;
      }
      // Eliminamos el "else if (!esIngreso) { ch -= importe }" para evitar el negativo doble.
    }
    // 3. BANCO: Suma ingresos, resta egresos (Correcto)
    // Incluye "Transferencia" (pagos de haberes) además de "Banco"
    else if (/banco|transferencia/i.test(row[4])) {
      bc += esIngreso ? importe : -importe;
    }
  }
  
  return { efectivo: ef, cheques: ch, banco: bc, total: ef + ch + bc };
}

  })).reverse();
  
  // Devolvemos el array directamente (sin envolver en {ok:true} para compatibilidad)
  return results; 
}


// ✅ NUEVA FUNCIÓN PARA EL DUEÑO (Rango de Fechas) - CORREGIDA ZONA HORARIA
function getMovimientosRango(startDateStr, endDateStr) {
  const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAMES.MOVIMIENTOS);
  if (!sh || sh.getLastRow() < 2) return { ok: true, data: [] };

  // Limpiamos los strings para quedarnos estrictamente con "YYYY-MM-DD"
  const startIso = startDateStr.split('T')[0];
  const endIso = endDateStr.split('T')[0];
  
  // 14 columnas para incluir Estado Cheque (col N) — lo usa la reconciliación del worker
  const dataRange = sh.getRange(2, 1, sh.getLastRow() - 1 || 1, 14);
  const data = dataRange.getValues();

  const results = data.filter(r => {
    if (!r[1]) return false;

    // Formateamos la fecha de la planilla a YYYY-MM-DD anclado a la hora de Argentina
    const fRowIso = Utilities.formatDate(new Date(r[1]), "America/Argentina/Buenos_Aires", "yyyy-MM-dd");

    // Comparamos los textos directamente (ej: "2026-03-27" <= "2026-03-27")
    return fRowIso >= startIso && fRowIso <= endIso;

  }).map(r => ({
    id: r[0],
    fecha: Utilities.formatDate(new Date(r[1]), "America/Argentina/Buenos_Aires", "yyyy-MM-dd"),
    hora: r[2] instanceof Date ? Utilities.formatDate(r[2], "America/Argentina/Buenos_Aires", "HH:mm") : String(r[2]),
    tipo: r[3], formaPago: r[4], banco: r[5], nro: r[6], importe: r[7], categoria: r[8], observacion: r[12],
    vehiculo: (String(r[12] || "").match(/\[Veh: (.*?)\]/i) || [])[1] || '',
    estado: String(r[13] || '')
  })).reverse();
  
  return { ok: true, data: results };
}


// Espejo de arqueo: NO recalcula el saldo — graba exactamente lo que
// Supabase ya decidió (fuente de verdad para arqueos desde 08/2026).
// Usa el mismo LockService que registrarArqueo por las dudas, pero acá
// es solo para evitar pisarse escribiendo la fila, no para "esperar
// el turno de recalcular" (no hay nada que recalcular).
function mirrorArqueo(p) {
  const lock = LockService.getScriptLock();
  try { lock.waitLock(10000); } catch (e) {
    logAccion(p.usuario, 'ERROR_MIRROR_ARQUEO', 'No se pudo tomar el lock', e.message);
    return { ok: false, error: 'Lock ocupado' };
  }
  try {
    const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAMES.ARQUEOS);
    const lastRow = sh.getLastRow();
    let nextId = 1;
    if (lastRow >= 2) {
      const lastIdVal = sh.getRange(lastRow, 1).getValue();
      nextId = (Number(lastIdVal) || 0) + 1;
    }
    const fisico = Number(p.efectivoFisico), sistema = Number(p.efectivoSistema), dif = Number(p.diferencia);
    const now = new Date();
    sh.appendRow([nextId, parseFechaSegura(p.fecha), p.usuario, fisico, sistema, dif, p.resultado, now, JSON.stringify(p.monitorData || {})]);
    SpreadsheetApp.flush();
    logAccion(p.usuario, 'ARQUEO (espejo Supabase)', `Físico: ${fisico} vs Sist: ${sistema}`, p.resultado);

    if (dif !== 0) {
      try {
        const tipo = dif > 0 ? 'Ingreso' : 'Egreso';
        registrarMovimientoCaja(tipo, 'Efectivo', Math.abs(dif), 'Ajuste Post-Arqueo', '', '', '', '', p.usuario, `Ajuste auto arqueo ${nextId}`, "", "", "");
      } catch (eAjuste) {
        logAccion(p.usuario, 'ERROR_AJUSTE_MIRROR_ARQUEO', `Arqueo #${nextId} espejado pero el ajuste de $${dif} FALLÓ`, eAjuste.message);
      }
    }
    return { ok: true, id: nextId };
  } catch (e) {
    logAccion(p.usuario, 'ERROR_FATAL_MIRROR_ARQUEO', e.message, e.stack);
    return { ok: false, error: e.message };
  } finally {
    lock.releaseLock();
  }
}

function registrarArqueo(user, fisico, monitorData) {
  // 🔒 1. PROTECCIÓN CONTRA DOBLE CLIC (LOCK SERVICE)
  // Esto obliga al script a esperar si hay otro arqueo guardándose en ese preciso milisegundo.
  const lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000); // Espera hasta 10 segundos para obtener turno
  } catch (e) {
    return { ok: false, error: "El sistema está ocupado guardando otro dato. Intenta de nuevo." };
  }

  try {
    const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAMES.ARQUEOS);
    
    // Recalculamos el estado de caja DENTRO del bloqueo para máxima precisión
    const estadoCaja = getEstadoCaja(); 
    const sist = estadoCaja.efectivo; // Usamos el efectivo calculado
    const dif = Number(fisico) - sist;
    const res = dif === 0 ? 'OK' : (dif > 0 ? 'Sobrante' : 'Faltante');
    const now = new Date();

    // 2. PROTECCIÓN DE MONITOR DATA
    // Si el JSON falla, no rompemos el arqueo, solo guardamos "{}"
    let monitorString = "{}";
    try {
      monitorString = JSON.stringify(monitorData || {});
    } catch (err) {
      console.log("Error parseando monitorData: " + err.message);
      monitorString = JSON.stringify({ error: "Data corrupta" });
    }

    analizarComportamiento(monitorData, "ARQUEO", user, dif);

    // 3. ID ROBUSTO
    // Si falla la lectura del último ID, asumimos que es el 1 o usamos Timestamp para no duplicar
    const lastRow = sh.getLastRow();
    let nextId = 1;
    if (lastRow >= 2) {
      const lastIdVal = sh.getRange(lastRow, 1).getValue();
      nextId = (Number(lastIdVal) || 0) + 1;
    }

    // Guardado en Sheet
    sh.appendRow([nextId, now, user, Number(fisico), sist, dif, res, now, monitorString]);
    
    // Forzamos que Google Sheets guarde los cambios AHORA MISMO
    SpreadsheetApp.flush(); 

    logAccion(user, 'ARQUEO', `Físico: ${fisico} vs Sist: ${sist}`, res);

    // Ajuste de Caja automático
    // (try/catch propio: si esto llegara a fallar, el arqueo en sí ya quedó
    // guardado arriba — no queremos perder ESE registro, pero sí necesitamos
    // que la falla quede visible en Logs en vez de desaparecer en silencio)
    if (dif !== 0) {
      try {
        const tipo = dif > 0 ? 'Ingreso' : 'Egreso';
        registrarMovimientoCaja(
          tipo, 'Efectivo', Math.abs(dif), 'Ajuste Post-Arqueo',
          '', '', '', '', user, `Ajuste auto arqueo ${nextId}`, "", "", ""
        );
      } catch (eAjuste) {
        logAccion(user, 'ERROR_AJUSTE_ARQUEO', `Arqueo #${nextId} guardado pero el ajuste de $${dif} FALLÓ`, eAjuste.message);
      }
    }

    if (Math.abs(dif) >= 20000) {
      registrarNotificacion('ANOMALIA_CAJA','ALTA','Desajuste Grande en Arqueo', `Faltan/Sobran $${dif}`, {usuario: user, fisico, sist});
    }

    // Auditoría IA
    try {
       analizarPatronesConIA(user, fisico, sist, monitorData);
    } catch(e) {
       console.log("⚠️ Error secundario en Auditoría IA: " + e.message);
    }

    return { ok: true, diferencia: dif }; // Devolvemos ok: true para que el frontend sepa que salió bien

  } catch (e) {
    // Si ocurre un error fatal, lo registramos en logs para que sepas qué pasó
    logAccion(user, 'ERROR_FATAL_ARQUEO', e.message, e.stack);
    return { ok: false, error: "Error interno al guardar: " + e.message };
  } finally {
    // 🔓 SIEMPRE liberamos el candado, pase lo que pase
    lock.releaseLock();
  }
}

function setupSheet(ss, name, headers, color) {
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  if (sh.getLastRow() === 0) {
    sh.appendRow(headers);
    sh.getRange(1,1,1,headers.length).setBackground(color).setFontColor('#FFF').setFontWeight('bold');
    sh.setFrozenRows(1);
  }
}

function getNotificacionesCaja(soloActivas) {
  const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAMES.NOTIFICACIONES);
  if (!sh) return [];
  const data = sh.getDataRange().getValues();
  const res = [];
  for (let i = 1; i < data.length; i++) {
    const r = data[i];
    if (!r[0]) continue;
    const obj = { id: r[0], fecha: r[1], hora: r[2], tipo: r[3], severidad: r[4], titulo: r[5], mensaje: r[6], contexto: r[7]?JSON.parse(r[7]):null, leida: r[8], borrada: r[9] };
    if (soloActivas && obj.borrada) continue;
    res.push(obj);
  }
  return res.sort((a,b)=> (new Date(b.fecha).getTime() - new Date(a.fecha).getTime()));
}

function getResumenBancosDueño() {
  const hoy = new Date();
  const fechaStr = Utilities.formatDate(hoy, "America/Argentina/Buenos_Aires", "yyyy-MM-dd");
  const mesCarpetaNombre = Utilities.formatDate(hoy, "en", "MMMM yyyy");
  
  let totalDia = 0;
  let detalles = [];

  try {
    const rootFolder = DriveApp.getFolderById(RECIBOS_FOLDER_ID);
    let targetFolder = rootFolder;

    // 1. Entrar a carpeta del Mes
    const subFolders = rootFolder.getFoldersByName(mesCarpetaNombre);
    if (subFolders.hasNext()) targetFolder = subFolders.next();

    const files = targetFolder.getFiles();
    
    // 2. Buscar archivos
    while (files.hasNext()) {
      const file = files.next();
      const name = file.getName();
      
      // Buscamos archivos de HOY (por nombre de archivo)
      if (name.includes(fechaStr) && !file.isTrashed()) {
         if (file.getMimeType() === MimeType.GOOGLE_SHEETS) {
            
            // 🔥 LLAMAMOS A LA NUEVA FUNCIÓN DE LECTURA FILA X FILA
            const infoArchivo = sumarTransferenciasFilaPorFila(file.getId());
            
            if (infoArchivo.total > 0) {
              totalDia += infoArchivo.total;
              // Guardamos un detalle limpio (ej: "Nico Turno Mañana")
              let nombreCorto = name.replace("Rendición de", "").replace(fechaStr, "").replace(".xlsx", "").trim();
              detalles.push({ archivo: nombreCorto, monto: infoArchivo.total });
            }
         }
      }
    }

    // 3. Calcular Acumulado Mensual (Suma todo el mes)
    const totalMes = calcularAcumuladoMensualFilas(targetFolder);

    return { ok: true, totalDia: totalDia, totalMes: totalMes, detalles: detalles };

  } catch (e) {
    return { ok: false, error: e.message };
  }
}

// 🔥 FUNCIÓN DE LECTURA EXACTA PARA TU PLANILLA
function sumarTransferenciasFilaPorFila(fileId) {
  try {
    const ss = SpreadsheetApp.openById(fileId);
    const sh = ss.getSheets()[0];
    // Obtenemos todos los datos de la hoja
    const data = sh.getDataRange().getValues();
    
    let suma = 0;

    // Recorremos desde la fila 1 (saltando encabezados si la fila 0 son titulos)
    // SEGÚN TU CSV:
    // Columna index 3 (D) = "Formas de Pago" (ej: "Transferencia: 61496.00")
    // Columna index 4 (E) = "Total Pagado" (ej: 61496)

    for (let i = 1; i < data.length; i++) {
      const row = data[i];
      if (row.length < 5) continue; // Seguridad para filas vacías

      const formaPago = String(row[3]).toUpperCase(); // Columna D
      const montoPagado = parseFloat(row[4]);         // Columna E (Total Pagado)

      // Palabras clave que indican que NO es efectivo
      if (formaPago.includes("TRANSFERENCIA") || 
          formaPago.includes("MERCADO") || 
          formaPago.includes("BANCO") || 
          formaPago.includes("ALIAS") ||
          formaPago.includes("CUENTA DNI")) {

          // Si es un número válido, lo sumamos
          if (!isNaN(montoPagado) && montoPagado > 0) {
            suma += montoPagado;
          }
      }
    }
    
    return { total: suma };

  } catch (e) {
    console.log("Error leyendo archivo: " + e.message);
    return { total: 0 };
  }
}

// Helper para sumar todo el mes con la misma lógica
function calcularAcumuladoMensualFilas(folder) {
  const files = folder.getFiles();
  let sumaMes = 0;
  while (files.hasNext()) {
    const file = files.next();
    if (file.getMimeType() === MimeType.GOOGLE_SHEETS && !file.isTrashed()) {
      sumaMes += sumarTransferenciasFilaPorFila(file.getId()).total;
    }
  }
  return sumaMes;
}

// 1. Detecta qué vehículo es basándose en la observación o categoría
function detectarVehiculo(observacion, categoria) {
  const obs = String(observacion || "").toLowerCase();
  const cat = String(categoria || "").toLowerCase();

  const match = obs.match(/\[veh:\s*(.*?)\]/i);
  if (match && match[1]) return match[1].trim().toUpperCase();

  if (obs.includes("kangoo") || cat.includes("kangoo")) return "KANGOO";
  if (obs.includes("saveiro") || cat.includes("saveiro")) return "SAVEIRO";
  if (obs.includes("hiace") || obs.includes("toyota") || cat.includes("toyota")) return "TOYOTA HIACE";
  if (obs.includes("uno") || obs.includes("fiat") || cat.includes("fiat")) return "FIAT UNO";
  if (obs.includes("partner") || cat.includes("partner")) return "PARTNER";
  
  return "FLOTA GENERAL";
}

// 2. Borra y reescribe la hoja LogsCombustible con datos limpios.
// Ya NO se llama en cada carga — se dispara sola cada 10 min via trigger
// (ver JOB_ActualizarTableroCombustiblePeriodico + setupTriggerTableroCombustible
// más abajo). Así el registro de un combustible cuesta lo mismo que cualquier
// otro egreso, y el tablero igual se mantiene al día (con hasta 10 min de demora).
function actualizarTableroCombustible() {
  const ss = SpreadsheetApp.getActive();
  const sourceSh = ss.getSheetByName(SHEET_NAMES.MOVIMIENTOS);
  let targetSh = ss.getSheetByName(SHEET_NAMES.LOGS_COMBUSTIBLE);
  if (!targetSh) targetSh = ss.insertSheet(SHEET_NAMES.LOGS_COMBUSTIBLE);
  
  const data = sourceSh.getDataRange().getValues();
  const logsLimpios = [];

  for (let i = 1; i < data.length; i++) {
    const row = data[i];
    const tipo = String(row[3]);
    const categoria = String(row[8]);
    
    if (tipo === 'Egreso' && categoria.toLowerCase().includes('combustible')) {
      const fecha = row[1];
      const importe = Number(row[7]);
      const usuario = row[11];
      const observacion = row[12];
      const vehiculo = detectarVehiculo(observacion, categoria);
      const mesAnio = Utilities.formatDate(new Date(fecha), "America/Argentina/Buenos_Aires", "MM/yyyy");

      logsLimpios.push([new Date(fecha), vehiculo, importe, usuario, mesAnio, observacion]);
    }
  }

  // Ordenar por fecha descendente (más nuevo arriba)
  logsLimpios.sort((a, b) => b[0] - a[0]);

  targetSh.clearContents(); 
  targetSh.appendRow(['FECHA', 'VEHÍCULO', 'MONTO ($)', 'RESPONSABLE', 'PERÍODO', 'DETALLE']);
  
  if (logsLimpios.length > 0) {
    targetSh.getRange(2, 1, logsLimpios.length, 6).setValues(logsLimpios);
  }

  // Formato visual
  targetSh.getRange("A1:F1").setBackground("#37474F").setFontColor("white").setFontWeight("bold");
  targetSh.getRange(2, 1, targetSh.getLastRow(), 1).setNumberFormat("dd/MM/yyyy");
  targetSh.getRange(2, 3, targetSh.getLastRow(), 1).setNumberFormat("$ #,##0.00");
  targetSh.autoResizeColumns(1, 6);
}

// 2.1 Job periódico: refresca el tablero de combustible fuera del camino
// crítico del registro. Se instala UNA vez con setupTriggerTableroCombustible().
function JOB_ActualizarTableroCombustiblePeriodico() {
  try { actualizarTableroCombustible(); }
  catch(e) { console.log('Error actualizando tablero combustible: ' + e.message); }
}

// Ejecutar UNA sola vez (manualmente desde el editor, o via clasp run) para
// instalar el trigger de 10 minutos. Es seguro volver a ejecutarla: borra
// cualquier trigger previo de esta misma función antes de crear uno nuevo,
// así nunca queda duplicado.
function setupTriggerTableroCombustible() {
  ScriptApp.getProjectTriggers().forEach(t => {
    if (t.getHandlerFunction() === 'JOB_ActualizarTableroCombustiblePeriodico') {
      ScriptApp.deleteTrigger(t);
    }
  });
  ScriptApp.newTrigger('JOB_ActualizarTableroCombustiblePeriodico')
    .timeBased()
    .everyMinutes(10)
    .create();
  return { ok: true, mensaje: 'Trigger instalado: cada 10 minutos' };
}

/****************************************************

 * Cruza fuentes independientes (rendiciones de Drive)
 * contra lo que Laura declaró en el arqueo del día.

function getResumenCruceDiario(dateStr) {
  const ZONA_AR = "America/Argentina/Buenos_Aires";
  const fechaBusq = parseFechaSegura(dateStr);
  const fStr = Utilities.formatDate(fechaBusq, ZONA_AR, "yyyy-MM-dd");
  const ss = SpreadsheetApp.getActive();

  // ---- 1. RENDICIONES DEL DÍA (fuente independiente: vienen de Drive) ----
  const rendSh = ss.getSheetByName(SHEET_NAMES.RENDICIONES);
  const rendiciones = [];
  let totalRendContado = 0;
  let totalRendEsperado = 0;

  if (rendSh && rendSh.getLastRow() >= 2) {
    rendSh.getDataRange().getValues().slice(1).forEach(row => {
      if (!row[1]) return;
      const fRow = Utilities.formatDate(new Date(row[1]), ZONA_AR, "yyyy-MM-dd");
      if (fRow !== fStr) return;
      const esperado = Number(row[4]) || 0;
      const contado  = Number(row[5]) || 0;
      rendiciones.push({ turno: row[2], repartidor: row[3], esperado, contado, tipo: row[7] });
      totalRendEsperado += esperado;
      totalRendContado  += contado;
    });
  }

  // ---- 2. MOVIMIENTOS MANUALES DEL DÍA (excluyendo los auto-generados) ----
  // Los auto-generados son: bases de rendición, ajustes de rendición, ajustes de arqueo.
  // Queremos solo lo que Laura registró manualmente.
  const CATEGORIAS_AUTO = [
    'rendición reparto - base',
    'diferencia rendición - ajuste',
    'ajuste post-arqueo'
  ];
  const movSh = ss.getSheetByName(SHEET_NAMES.MOVIMIENTOS);
  let ingresosManual = 0;
  let egresosManual  = 0;
  const movDetalle   = [];

  if (movSh && movSh.getLastRow() >= 2) {
    movSh.getDataRange().getValues().slice(1).forEach(row => {
      if (!row[1] || !/efectivo/i.test(row[4])) return;
      const fRow = Utilities.formatDate(new Date(row[1]), ZONA_AR, "yyyy-MM-dd");
      if (fRow !== fStr) return;
      const cat = String(row[8] || "").toLowerCase();
      if (CATEGORIAS_AUTO.some(c => cat.includes(c))) return; // saltar auto-generados
      const importe = Number(row[7]) || 0;
      if (row[3] === 'Ingreso') ingresosManual += importe;
      else if (row[3] === 'Egreso') egresosManual += importe;

    });
  }

  // ---- 3. ARQUEO DEL DÍA ----
  const arqueoSh = ss.getSheetByName(SHEET_NAMES.ARQUEOS);
  let arqueo = null;
  if (arqueoSh && arqueoSh.getLastRow() >= 2) {

    for (let i = rows.length - 1; i >= 0; i--) {

      const fRow = Utilities.formatDate(new Date(rows[i][1]), ZONA_AR, "yyyy-MM-dd");

          fisico:     Number(rows[i][3]),
          sistema:    Number(rows[i][4]),
          diferencia: Number(rows[i][5]),
          resultado:  rows[i][6],
          hora:       rows[i][7] instanceof Date
                        ? Utilities.formatDate(rows[i][7], ZONA_AR, "HH:mm")
                        : String(rows[i][7])
        };
        break;
      }
    }
  }

  // ---- 4. SALDO APERTURA = arqueo del día anterior ----
  let saldoApertura = 0;
  if (arqueoSh && arqueoSh.getLastRow() >= 2) {
    const ayer    = new Date(fechaBusq.getTime() - 86400000);
    const ayerStr = Utilities.formatDate(ayer, ZONA_AR, "yyyy-MM-dd");
    const rows    = arqueoSh.getDataRange().getValues().slice(1);
    for (let i = rows.length - 1; i >= 0; i--) {
      if (!rows[i][1]) continue;
      const fRow = Utilities.formatDate(new Date(rows[i][1]), ZONA_AR, "yyyy-MM-dd");
      if (fRow === ayerStr) {
        // Usamos el saldo sistema (ya ajustado por el arqueo de ayer)
        saldoApertura = Number(rows[i][4]);
        break;
      }
    }
  }

  // ---- 5. CRUCE FINAL ----
  // Qué debería haber en caja al cerrar el día:
  // apertura + lo que entró de rendiciones (independiente) + ingresos manuales - egresos manuales
  const saldoEsperado = saldoApertura + totalRendContado + ingresosManual - egresosManual;

  // Brecha = lo que Laura declaró vs lo que el cruce espera
  // Negativo = declaró menos de lo esperado (posible robo)
  // Positivo = declaró más (raro, podría ser ingreso no categorizado)
  const brecha = arqueo ? (arqueo.fisico - saldoEsperado) : null;
  const alerta = brecha !== null && Math.abs(brecha) >= 1000;

  // Si hay alerta, crear notificación para Pablo
  if (alerta) {
    registrarNotificacion(
      'CRUCE_RENDICION_ARQUEO',
      Math.abs(brecha) >= 5000 ? 'ALTA' : 'MEDIA',
      `Brecha detectada el ${fStr}`,
      `El cruce de rendiciones vs arqueo muestra una diferencia de $${brecha}. ` +
      `Esperado: $${saldoEsperado} | Declarado: $${arqueo.fisico}`,
      { fecha: fStr, saldoEsperado, declarado: arqueo ? arqueo.fisico : null, brecha }
    );
  }

  return {
    ok: true,
    fecha: fStr,
    saldoApertura,
    rendiciones: {
      detalle: rendiciones,
      totalEsperado: totalRendEsperado,
      totalContado:  totalRendContado,
      diferencia:    totalRendContado - totalRendEsperado
    },
    movimientosManuales: {
      ingresos: ingresosManual,
      egresos:  egresosManual,
      neto:     ingresosManual - egresosManual,
      detalle:  movDetalle
    },
    saldoEsperado,
    arqueo,
    brecha,
    alerta,
    sinArqueo:     !arqueo,
    sinRendiciones: rendiciones.length === 0
  };
}


// Versión por rango de fechas (para el panel de tendencias de Pablo)
function getResumenCruceRango(startDateStr, endDateStr) {
  const ZONA_AR = "America/Argentina/Buenos_Aires";
  const start = new Date(startDateStr.split('T')[0]);
  const end   = new Date(endDateStr.split('T')[0]);
  const resultados = [];

  // Iteramos día por día en el rango
  const cur = new Date(start);
  while (cur <= end) {
    const fStr = Utilities.formatDate(cur, ZONA_AR, "yyyy-MM-dd");
    const resumen = getResumenCruceDiario(fStr);
    // Solo incluimos días con arqueo registrado
    if (!resumen.sinArqueo) {
      resultados.push({
        fecha:          resumen.fecha,
        saldoApertura:  resumen.saldoApertura,
        rendContado:    resumen.rendiciones.totalContado,
        ingresosManual: resumen.movimientosManuales.ingresos,
        egresosManual:  resumen.movimientosManuales.egresos,
        saldoEsperado:  resumen.saldoEsperado,
        declarado:      resumen.arqueo ? resumen.arqueo.fisico : null,
        brecha:         resumen.brecha,
        alerta:         resumen.alerta
      });
    }
    cur.setDate(cur.getDate() + 1);
  }

  return { ok: true, data: resultados };
}

/**
 * Busca una rendición específica en Supabase por fecha + turno + repartidor.
 */
function buscarRendicionEnSupabase_(fecha, turno, repartidor) {
  try {
    var params = 'fecha=eq.' + fecha
               + '&turno=eq.'       + encodeURIComponent(turno)
               + '&repartidor=eq.'  + encodeURIComponent(repartidor)
               + '&limit=1';

    var resp = UrlFetchApp.fetch(
      SB_REND_URL + '/rest/v1/rendiciones_reparto?' + params,
      { method: 'GET',
        headers: { 'apikey': SB_REND_KEY, 'Authorization': 'Bearer ' + SB_REND_KEY },
        muteHttpExceptions: true }
    );
    var rows = JSON.parse(resp.getContentText());
    return (rows && rows.length > 0) ? rows[0] : null;
  } catch (e) {
    Logger.log('❌ buscarRendicionEnSupabase_: ' + e.message);
    return null;
  }
}

// REPORTE MENSUAL COMPARATIVO (1ro de cada mes, 08:00)

function JOB_ReporteRepartoMensual() {
  var EMAIL = "santamariapablodaniel@gmail.com";
  var ZONA  = "America/Argentina/Buenos_Aires";
  
  var hoy          = new Date();
  var mesActual    = new Date(hoy.getFullYear(), hoy.getMonth() - 1, 1);   // mes que acaba de cerrar
  var mesAnterior  = new Date(hoy.getFullYear(), hoy.getMonth() - 2, 1);   // mes previo para comparar
  
  function fetchMes(inicio) {
    var fin = new Date(inicio.getFullYear(), inicio.getMonth() + 1, 0);
    var params = 'fecha=gte.' + Utilities.formatDate(inicio, ZONA, 'yyyy-MM-dd')
               + '&fecha=lte.' + Utilities.formatDate(fin,   ZONA, 'yyyy-MM-dd')
               + '&procesado=eq.true';

      method: 'GET',
      headers: { 'apikey': SB_REND_KEY, 'Authorization': 'Bearer ' + SB_REND_KEY },
      muteHttpExceptions: true
    });
    return JSON.parse(resp.getContentText()) || [];
  }
  
  function resumir(rows) {
    var r = { turnos:0, entregas:0, devoluciones:0, noPago:0, ef:0, tr:0, ch:0, total:0, duraciones:[] };
    rows.forEach(function(row) {
      var detalle = row.detalle_clientes || [];

      r.entregas     += detalle.filter(function(c){ return !c.devolucion && c.formaPago1 !== 'No pagó'; }).length;

      r.ef           += row.efectivo || 0;
      r.tr           += row.transferencia || 0;
      r.ch           += row.cheque || 0;
      r.total        += row.total || 0;
      var tss = detalle.map(function(c){ return c.ts ? new Date(c.ts).getTime() : null; }).filter(Boolean);
      if (tss.length > 1) r.duraciones.push((Math.max.apply(null,tss) - Math.min.apply(null,tss)) / 60000);
    });
    r.promEntregas   = r.turnos > 0 ? (r.entregas / r.turnos).toFixed(1) : 0;
    r.promDuracion   = r.duraciones.length > 0 ? Math.round(r.duraciones.reduce(function(a,b){return a+b;},0) / r.duraciones.length) : null;
    return r;
  }
  
  var meses_es = ['Enero','Febrero','Marzo','Abril','Mayo','Junio','Julio','Agosto','Septiembre','Octubre','Noviembre','Diciembre'];
  var dataMes  = resumir(fetchMes(mesActual));
  var dataAnt  = resumir(fetchMes(mesAnterior));
  var labelMes = meses_es[mesActual.getMonth()]  + ' ' + mesActual.getFullYear();
  var labelAnt = meses_es[mesAnterior.getMonth()] + ' ' + mesAnterior.getFullYear();
  var fmt = function(n){ return '$ ' + Number(n).toLocaleString('es-AR'); };
  
  function delta(a, b) {
    if (!b) return '';
    var d = ((a - b) / b * 100).toFixed(1);
    var color = d >= 0 ? '#0d9f57' : '#d4302f';
    return ' <span style="font-size:11px;color:' + color + '">(' + (d >= 0 ? '+' : '') + d + '%)</span>';
  }
  
  function fila(label, valMes, valAnt, fmt) {
    return '<tr style="border-bottom:1px solid #f0f0f0">'
      + '<td style="padding:10px 14px;font-size:13px;color:#555">' + label + '</td>'
      + '<td style="padding:10px 14px;text-align:right;font-weight:700;font-size:14px">' + (fmt ? fmt(valMes) : valMes) + delta(valMes, valAnt) + '</td>'
      + '<td style="padding:10px 14px;text-align:right;font-size:13px;color:#999">' + (fmt ? fmt(valAnt) : valAnt) + '</td>'
      + '</tr>';
  }
  
  var html = '<!DOCTYPE html><html><body style="font-family:sans-serif;background:#f5f7fa;padding:20px">'
    + '<div style="max-width:600px;margin:0 auto;background:#fff;border-radius:12px;overflow:hidden;box-shadow:0 4px 20px rgba(0,0,0,.08)">'
    + '<div style="background:linear-gradient(135deg,#0d9f57,#10b981);padding:30px;color:#fff;text-align:center">'
    + '<h1 style="margin:0;font-size:22px">📊 Reporte Mensual de Reparto</h1>'
    + '<p style="margin:8px 0 0;opacity:.85;font-size:14px">' + labelMes + ' vs ' + labelAnt + '</p></div>'
    + '<div style="padding:24px">'
    + '<table style="width:100%;border-collapse:collapse">'
    + '<thead><tr><th style="padding:8px 14px;text-align:left;font-size:11px;color:#999;text-transform:uppercase">Métrica</th>'
    + '<th style="padding:8px 14px;text-align:right;font-size:11px;color:#2563eb;text-transform:uppercase">' + labelMes + '</th>'
    + '<th style="padding:8px 14px;text-align:right;font-size:11px;color:#999;text-transform:uppercase">' + labelAnt + '</th>'
    + '</tr></thead><tbody>'
    + fila('Turnos trabajados',       dataMes.turnos,       dataAnt.turnos)
    + fila('Total entregas',          dataMes.entregas,     dataAnt.entregas)
    + fila('Promedio entregas/turno', dataMes.promEntregas, dataAnt.promEntregas)
    + fila('Devoluciones',            dataMes.devoluciones, dataAnt.devoluciones)
    + fila('Sin cobrar',              dataMes.noPago,       dataAnt.noPago)
    + fila('Duración prom. de turno', dataMes.promDuracion !== null ? dataMes.promDuracion + ' min' : '—',
                                      dataAnt.promDuracion !== null ? dataAnt.promDuracion + ' min' : '—')
    + fila('Recaudación total',       dataMes.total,        dataAnt.total,    fmt)
    + fila('Efectivo',                dataMes.ef,           dataAnt.ef,       fmt)
    + fila('Transferencia',           dataMes.tr,           dataAnt.tr,       fmt)
    + fila('Cheque',                  dataMes.ch,           dataAnt.ch,       fmt)
    + '</tbody></table></div>'
    + '<div style="background:#f8f9fa;text-align:center;padding:14px;font-size:12px;color:#90a4ae;border-top:1px solid #eee">Mercado Limpio · Analytics Automático</div>'
    + '</div></body></html>';
  
  MailApp.sendEmail({ to: EMAIL, subject: '📊 Reparto ' + labelMes + ' vs ' + labelAnt, htmlBody: html });
  Logger.log('✅ Reporte mensual enviado.');
}

function obtenerHistorialRendicionesHoja_() {
  const ss = SpreadsheetApp.getActive();
  const sh = ss.getSheetByName(SHEET_NAMES.RENDICIONES);
  if (!sh || sh.getLastRow() < 2) return [];

  const data = sh.getDataRange().getValues();
  const result = [];
  // Columnas: ID, Fecha, Turno, Repartidor, Esperado, Contado, Diferencia, TipoDif, Usuario, Timestamp
  for (let i = data.length - 1; i >= 1 && result.length < 40; i--) {
    const row = data[i];
    if (!row[1]) continue;
    result.push({
      id:         Number(row[0]),
      fecha:      Utilities.formatDate(new Date(row[1]), "America/Argentina/Buenos_Aires", "yyyy-MM-dd"),
      turno:      String(row[2] || ''),
      repartidor: String(row[3] || ''),
      esperado:   Number(row[4] || 0),
      contado:    Number(row[5] || 0),
      diferencia: Number(row[6] || 0),
      tipo:       String(row[7] || 'Exacto'),
      usuario:    String(row[8] || ''),
      ts:         row[9] ? Utilities.formatDate(new Date(row[9]), "America/Argentina/Buenos_Aires", "dd/MM HH:mm") : ''
    });
  }
  return result;
}


function editarMovimientoCaja(id, tipo, formaPago, importe, categoria, observacion) {
  const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAMES.MOVIMIENTOS);
  const lastRow = sh.getLastRow();
  if (lastRow < 2) return { ok: false, error: "Sin datos en la hoja" };
  const ids = sh.getRange(2, 1, lastRow - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) {
    if (Number(ids[i][0]) === Number(id)) {
      const row = i + 2;
      if (tipo        != null) sh.getRange(row, 4).setValue(tipo);
      if (formaPago   != null) sh.getRange(row, 5).setValue(formaPago);
      if (importe     != null) sh.getRange(row, 8).setValue(Number(importe));
      if (categoria   != null) sh.getRange(row, 9).setValue(categoria);
      if (observacion != null) sh.getRange(row, 13).setValue(observacion);
      logAccion("Pablo", 'EDITAR_MOVIMIENTO_CAJA', 'ID: ' + id, String(categoria || ''));
      return { ok: true };
    }
  }
  return { ok: false, error: 'Movimiento no encontrado (ID: ' + id + ')' };
}

function eliminarMovimientoCaja(id) {
  const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAMES.MOVIMIENTOS);
  const lastRow = sh.getLastRow();
  if (lastRow < 2) return { ok: false, error: "Sin datos en la hoja" };
  const ids = sh.getRange(2, 1, lastRow - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) {
    if (Number(ids[i][0]) === Number(id)) {
      sh.deleteRow(i + 2);
      logAccion("Pablo", 'ELIMINAR_MOVIMIENTO_CAJA', 'ID: ' + id, '');
      return { ok: true };
    }
  }
  return { ok: false, error: 'Movimiento no encontrado (ID: ' + id + ')' };
}

function renombrarEmpleado(params) {
  const ss    = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName('ConfigEmpleados');
  const data  = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if ((data[i][0] || '').toString().trim() === (params.oldNombre || '').trim()) {
      sheet.getRange(i + 1, 1).setValue(params.nombre);
      return { ok: true };
    }
  }
  return { ok: false, error: 'Empleado no encontrado: ' + params.oldNombre };
}

function eliminarMovimientoPorCampos(params) {
  const ZONA     = "America/Argentina/Buenos_Aires";
  const fecha    = params.fecha;
  const hora     = (params.hora || "").substring(0, 5);
  const importe  = Number(params.importe);
  const categoria = (params.categoria || "").toLowerCase();

  const sheet = SpreadsheetApp.getActive().getSheetByName("MovimientosCaja");
  const data  = sheet.getDataRange().getValues();

  for (let i = 1; i < data.length; i++) {
    const rawFecha   = data[i][1];  // col B — Date object en GAS
    const rawHora    = data[i][2];  // col C — Date object en GAS
    const rowImporte = Number(data[i][7]);
    const rowCat     = (data[i][8] || "").toLowerCase();

    const rowFechaStr = rawFecha instanceof Date
      ? Utilities.formatDate(rawFecha, ZONA, "yyyy-MM-dd")
      : String(rawFecha).substring(0, 10);

    const rowHoraStr = rawHora instanceof Date
      ? Utilities.formatDate(rawHora, ZONA, "HH:mm")
      : String(rawHora).substring(0, 5);

    if (rowFechaStr === fecha && rowHoraStr === hora
        && rowImporte === importe && rowCat === categoria) {
      sheet.deleteRow(i + 1);
      return { ok: true, filaEliminada: i + 1 };
    }
  }
  return { ok: false, error: "No encontrado" };
}
